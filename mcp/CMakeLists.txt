# Independent MCP project: cmake -S mcp -B build-mcp.
# Source + tests + demos are checked into mcp/; never fetch myMcpTools.
cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(myMcpToolsIntegrated VERSION 0.2.0 LANGUAGES CXX)
  set(_mcp_standalone ON)
else()
  set(_mcp_standalone OFF)
endif()
option(MYAGENT_BUILD_MCP_EXAMPLES "Build standalone MCP demos" ${_mcp_standalone})
option(MYAGENT_BUILD_MCP_TESTS "Build all imported MCP regression tests" ${_mcp_standalone})
if(MYAGENT_BUILD_MCP_TESTS AND NOT MYAGENT_BUILD_MCP_EXAMPLES)
  message(FATAL_ERROR "MCP integration tests need the MCP examples; enable MYAGENT_BUILD_MCP_EXAMPLES.")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  if(MYAGENT_BUILD_MCP_TESTS)
    enable_testing()
  endif()
endif()
add_subdirectory(sdk)
if(MYAGENT_BUILD_MCP_TESTS)
  find_package(Python3 COMPONENTS Interpreter QUIET)
endif()
if(MYAGENT_BUILD_MCP_EXAMPLES)
  add_subdirectory(examples)
endif()
if(MYAGENT_BUILD_MCP_TESTS)
  add_subdirectory(tests)
endif()
