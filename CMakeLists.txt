cmake_minimum_required(VERSION 3.20)
project(myAgentConsole VERSION 0.2.0 LANGUAGES CXX)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MYAGENT_BUILD_CONSOLE "Build FTXUI console" ON)
option(MYAGENT_BUILD_EXAMPLES "Build Agent demos" OFF)
option(MYAGENT_BUILD_TESTS "Build Agent tests" OFF)
option(MYAGENT_CONSOLE_BUILD_TESTS "Build console tests" ON)
option(MYAGENT_BUILD_MCP_EXAMPLES "Build MCP demos" OFF)
option(MYAGENT_BUILD_MCP_TESTS "Build MCP tests" OFF)
# 可选的 23 个共享 DeveloperTools，Agent / MCP 注册使用同一套 Handler。
option(MYAGENT_BUILD_DEVELOPER_TOOLS "Build optional native DeveloperTools library" ON)
option(MYAGENT_BUILD_DEVELOPER_TOOLS_MCP "Build standalone DeveloperTools MCP stdio server" ON)
option(MYAGENT_BUILD_DEVELOPER_TOOLS_TESTS "Build offline DeveloperTools tests" ON)
if(MYAGENT_BUILD_TESTS OR MYAGENT_BUILD_MCP_TESTS OR
   (MYAGENT_BUILD_CONSOLE AND MYAGENT_CONSOLE_BUILD_TESTS) OR
   (MYAGENT_BUILD_DEVELOPER_TOOLS AND MYAGENT_BUILD_DEVELOPER_TOOLS_TESTS))
  enable_testing()
endif()
add_subdirectory(mcp/sdk)
add_subdirectory(sdk)
if(MYAGENT_BUILD_DEVELOPER_TOOLS)
  add_subdirectory(tools/developer_tools)
  if(MYAGENT_BUILD_DEVELOPER_TOOLS_MCP)
    add_subdirectory(mcp_servers/developer_tools)
  endif()
endif()
if(MYAGENT_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
if(MYAGENT_BUILD_MCP_EXAMPLES)
  add_subdirectory(mcp/examples)
endif()

if(MYAGENT_BUILD_TESTS)
  add_subdirectory(tests)
endif()
if(MYAGENT_BUILD_MCP_TESTS)
  add_subdirectory(mcp/tests)
endif()
if(MYAGENT_BUILD_CONSOLE)
  add_subdirectory(console)
endif()

# Organize independent demo projects in Visual Studio's Solution Explorer.
if(MYAGENT_BUILD_EXAMPLES)
  get_property(_agent_examples DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/examples" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(_target IN LISTS _agent_examples)
    set_property(TARGET "${_target}" PROPERTY FOLDER "示例/Agent")
  endforeach()
endif()
if(MYAGENT_BUILD_MCP_EXAMPLES)
  get_property(_mcp_examples DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/mcp/examples" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(_target IN LISTS _mcp_examples)
    set_property(TARGET "${_target}" PROPERTY FOLDER "示例/MCP")
  endforeach()
endif()

if(MYAGENT_BUILD_TESTS)
  get_property(_agent_tests DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/tests" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(_target IN LISTS _agent_tests)
    set_property(TARGET "${_target}" PROPERTY FOLDER "测试/Agent")
  endforeach()
endif()
if(MYAGENT_BUILD_MCP_TESTS)
  get_property(_mcp_tests DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/mcp/tests" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(_target IN LISTS _mcp_tests)
    set_property(TARGET "${_target}" PROPERTY FOLDER "测试/MCP")
  endforeach()
endif()
foreach(_target IN ITEMS myagent_console_commands_tests myagent_console_chat_tests developer_tools_tests)
  if(TARGET "${_target}")
    set_property(TARGET "${_target}" PROPERTY FOLDER "测试/控制台和工具")
  endif()
endforeach()
foreach(_target IN ITEMS screen dom component yaml-cpp)
  if(TARGET "${_target}")
    set_property(TARGET "${_target}" PROPERTY FOLDER "依赖/第三方")
  endif()
endforeach()
foreach(_target IN ITEMS myagent mymcp myagent_console_session developer_tools_agent developer_tools_core myagent_developer_tools_mcp)
  if(TARGET "${_target}")
    set_property(TARGET "${_target}" PROPERTY FOLDER "依赖/项目组件")
  endif()
endforeach()
