cmake_minimum_required(VERSION 3.20)
project(myAgentConsole VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MYAGENT_BUILD_CONSOLE "Build FTXUI console" ON)
option(MYAGENT_BUILD_EXAMPLES "Build Agent demos" OFF)
option(MYAGENT_BUILD_TESTS "Build Agent tests" OFF)
option(MYAGENT_CONSOLE_BUILD_TESTS "Build console tests" ON)
option(MYAGENT_BUILD_MCP_EXAMPLES "Build MCP demos" OFF)
option(MYAGENT_BUILD_MCP_TESTS "Build MCP tests" OFF)
# 可选的 23 个共享 DeveloperTools，Agent / MCP 注册使用同一套 Handler。
option(MYAGENT_BUILD_DEVELOPER_TOOLS "Build optional native DeveloperTools library" ON)
option(MYAGENT_BUILD_DEVELOPER_TOOLS_MCP "Build standalone DeveloperTools MCP stdio server" ON)
option(MYAGENT_BUILD_DEVELOPER_TOOLS_TESTS "Build offline DeveloperTools tests" ON)
if(MYAGENT_BUILD_TESTS OR MYAGENT_BUILD_MCP_TESTS OR
   (MYAGENT_BUILD_CONSOLE AND MYAGENT_CONSOLE_BUILD_TESTS) OR
   (MYAGENT_BUILD_DEVELOPER_TOOLS AND MYAGENT_BUILD_DEVELOPER_TOOLS_TESTS))
  enable_testing()
endif()
add_subdirectory(mcp/sdk)
add_subdirectory(sdk)
if(MYAGENT_BUILD_DEVELOPER_TOOLS)
  add_subdirectory(tools/developer_tools)
  if(MYAGENT_BUILD_DEVELOPER_TOOLS_MCP)
    add_subdirectory(mcp_servers/developer_tools)
  endif()
endif()
if(MYAGENT_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
if(MYAGENT_BUILD_MCP_EXAMPLES)
  add_subdirectory(mcp/examples)
endif()
if(MYAGENT_BUILD_TESTS)
  add_subdirectory(tests)
endif()
if(MYAGENT_BUILD_MCP_TESTS)
  add_subdirectory(mcp/tests)
endif()
if(MYAGENT_BUILD_CONSOLE)
  add_subdirectory(console)
endif()
