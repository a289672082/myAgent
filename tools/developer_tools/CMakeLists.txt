# 共享执行库仅依赖 MCP Runtime；单独的 Agent adapter 才引用 myAgent。
add_library(developer_tools_core STATIC src/developer_tools.cpp)
target_compile_features(developer_tools_core PUBLIC cxx_std_20)
target_include_directories(developer_tools_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(developer_tools_core PUBLIC myagent::mcp)
add_library(developer_tools_agent STATIC src/agent_adapter.cpp)
target_compile_features(developer_tools_agent PUBLIC cxx_std_20)
target_link_libraries(developer_tools_agent PUBLIC developer_tools_core myagent::agent)
if(MYAGENT_BUILD_DEVELOPER_TOOLS_TESTS)
  add_executable(developer_tools_tests tests/test_developer_tools.cpp)
  target_link_libraries(developer_tools_tests PRIVATE developer_tools_agent)
  add_test(NAME developer_tools COMMAND developer_tools_tests)
  set_tests_properties(developer_tools PROPERTIES TIMEOUT 45)
endif()
